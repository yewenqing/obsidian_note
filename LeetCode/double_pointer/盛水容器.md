class Solution:

    def moveZeroes(self, nums: list[int]) -> None:
        """
        Do not return anything, modify nums in-place instead.
        """
        if len(nums) == 1 or len(nums) == 0:       \
            return
        l = len(nums)
        while 0 in nums:
            nums.remove(0)
        for i in range(l - len(nums)):
            nums.append(0)
