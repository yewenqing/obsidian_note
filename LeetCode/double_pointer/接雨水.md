class Solution:
    def trap(self, height: list[int]) -> int:
        if len(height) < 2:
            return 0
        L = 0
        R = len(height) - 1
        while height[L] == 0 and L < R:
            L += 1
        while height[R] == 0 and L < R:
            R -= 1            

        #现在找到了两侧的有效墙
        # 找到较低的一侧墙, 高度记为low，
        # 较低一侧墙向内缩，直到找到高于或平于原来的墙，期间计算low - curr height 并累加至total water
        water_level= 0
        total_water = 0
        while L < R:
            low = min(height[L], height[R])
    
            if height[L] < height[R]:
                while height[L] <= low and L < R:
                    total_water +=  low - height[L]
                    L += 1
            else:
 
                while height[R] <= low and L < R:
                    total_water +=  low - height[R]
                    R -= 1
        return total_water
