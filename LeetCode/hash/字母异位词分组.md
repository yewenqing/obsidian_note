class Solution:

    def groupAnagrams(self, strs: list[str]) -> list[list[str]]:

        hash_map = {}

        for i, s in enumerate(strs):

            key = key = "".join(sorted(s))

            if key in hash_map:

                hash_map[key].append(s)

            else:

                hash_map[key] = [s]

        return list(hash_map.values())
